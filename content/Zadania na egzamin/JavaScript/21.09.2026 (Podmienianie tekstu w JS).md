*index.html*
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Document</title>
</head>
<body>
    <button id="zmiana" onclick="zmiana()">Naciśnij w celu zmiany treści napisu poniżej...</button>
    <p id="stary">Stary napis...</p>

    <script>
	    // metoda inline (najszybsza)
        function zmiana() {
	        // metoda pobierania id
            let stary=document.getElementById("stary");
	        // metoda pobierania selektora
            let stary=document.querySelector("p");
            let stary=document.querySelector("#stary");

            // metoda dodawania HTML
            stary.innerHTML = "Nowy napis...";
            // metoda dodawania samego tekstu
            stary.textContent = "Nowy napis...";
        }
        
        // metoda addEventListener (współczesna, zewnętrzna)
        let baton=document.getElementById("baton");
        baton.addEventListener('click', zmiana);
        baton.onclick = zmiana;
    </script>
</body>
</html>
```
