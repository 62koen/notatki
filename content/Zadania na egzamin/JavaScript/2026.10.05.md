Załączniki są na dole strony.

*slider.html*
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Document</title>
</head>
<body>
    <aside id="left"><span onclick="poprzednie()">&lt;</span></aside>
    <div id="middle"><img src="obr1.jpg" height="400px" id="img1"><p id="nazwa">obr1.jpg</p></div>
    <aside id="right"><span onclick="nastepne()">&gt;</span></aside>
    <script>
        let i=1;
        let img = document.getElementById("img1");
        let nazwa = document.getElementById("nazwa");
        function poprzednie() {
            i--;
            if (i<1) {
                i=6;
            }
            img.src = 'obr' + i + '.jpg';
            nazwa.innerHTML = 'obr' + i + '.jpg';
        }
        function nastepne() {
            i++;
            if (i>6) {
                i=1;
            }
            img.src = 'obr' + i + '.jpg';
            nazwa.innerHTML = 'obr' + i + '.jpg';
        }
    </script>
</body>
</html>
```
